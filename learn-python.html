📌 条件分岐

# if / elif / else
score = 75
if score >= 80:
    print("優")
elif score >= 60:
    print("良")
else:
    print("要復習")

# 三項演算子（条件式）
result = "合格" if score >= 60 else "不合格"

# in / not in で所属チェック
if "a" in "apple":
    print("含む")

📌 ループ

# for ループ（range）
for i in range(5):
    print(i)   # 0,1,2,3,4

# リストを直接イテレート
fruits = ["apple", "banana", "cherry"]
for f in fruits:
    print(f)

# インデックス付きで回す
for i, f in enumerate(fruits):
    print(i, f)

# while ループ
count = 0
while count < 3:
    count += 1

# break / continue
for n in range(10):
    if n == 3:
        continue   # 3だけスキップ
    if n == 6:
        break      # 6でループ終了

📌 関数

# 関数定義
def greet(name, greeting="Hello"):
    # greeting はデフォルト引数
    return f"{greeting}, {name}!"

print(greet("World"))            # Hello, World!
print(greet("太郎", "こんにちは"))  # こんにちは, 太郎!

# 可変長引数
def total(*nums):
    return sum(nums)

print(total(1, 2, 3))          # 6

# lambda（無名関数）
square = lambda x: x * x
print(square(5))            # 25

📌 リスト・辞書内包表記

# リスト内包表記
squares = [x * x for x in range(10)]
evens = [x for x in range(10) if x % 2 == 0]

# 辞書内包表記
word_len = {w: len(w) for w in ["a", "bb", "ccc"]}
# {'a': 1, 'bb': 2, 'ccc': 3}

# 集合内包表記
uniq_lens = {len(w) for w in ["a", "bb", "cc"]}

📌 ファイル操作・例外処理

# with文でファイルを安全に開く（自動クローズ）
with open("data.txt", "r", encoding="utf-8") as f:
    for line in f:
        print(line.strip())

# 書き込み
with open("out.txt", "w", encoding="utf-8") as f:
    f.write("result\n")

# 例外処理
try:
    value = 10 / 0
except ZeroDivisionError as e:
    print(f"エラー: {e}")
finally:
    print("処理終了")
